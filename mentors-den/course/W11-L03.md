# Player
player = {
    "name": "Paw",

    "attack": 15,
    "defense": 5,
    "exp": 0,
    "level": 1,
}

# Enemies
enemies = [
    {"name": "Slime", "hp": 30, "attack": 5, "defense": 2, "exp": 10},
    {"name": "Wolf", "hp": 60, "attack": 10, "defense": 5, "exp": 25},
    {"name": "Dragon", "hp": 150, "attack": 25, "defense": 15, "exp": 100},
]

def pick_enemy():
    # Random enemy from list
    return random.choice(enemies)

def show_status(player, enemy):
    print(f"\n{player['name']} (Lv{player['level']}): HP {player['hp']}/{player['max_hp']}")
    print(f"{enemy['name']}: HP {enemy['hp']}")

## 三、战斗循环（15 分钟）

```python
def calc_dmg(attacker_atk, defender_def):
    base = attacker_atk - defender_def
    return max(1, base + random.randint(-2, 3))  # add randomness

def battle(player, enemy):
    print(f"\n=== Battle vs {enemy['name']}! ===")

    while enemy["hp"] > 0 and player["hp"] > 0:
        show_status(player, enemy)
        action = input("\n(attack/run): ").strip().lower()

        if action == "run":
            if random.random() < 0.5:
                print("Escaped!")
                return "ran"
            else:
                print("Failed to escape!")

        # Player attack
        dmg = calc_dmg(player["attack"], enemy["defense"])
        enemy["hp"] -= dmg
        print(f"You hit {enemy['name']} for {dmg} damage!")

        if enemy["hp"] <= 0:
            print(f"You defeated {enemy['name']}!")
            player["exp"] += enemy["exp"]
            check_levelup(player)
            return "won"

        # Enemy attack
        dmg = calc_dmg(enemy["attack"], player["defense"])
        player["hp"] -= dmg
        print(f"{enemy['name']} hits you for {dmg}!")

    if player["hp"] <= 0:
        print("You have fallen...")
        return "lost"
    return "won"

def check_levelup(player):
    needed = player["level"] * 50
    if player["exp"] >= needed:
        player["level"] += 1
        player["max_hp"] += 20
        player["hp"] = player["max_hp"]
        player["attack"] += 5
        player["defense"] += 2
        print(f"\n*** LEVEL UP! Now Lv{player['level']} ***")
        print(f"  HP: {player['max_hp']}  ATK: {player['attack']}  DEF: {player['defense']}")
```

### 拆解

- **`calc_dmg(atk, df)`**：伤害公式 + 随机 ±2~+3，让战斗不死板
- **`random.random() < 0.5`**：50% 逃跑成功率
- **`check_levelup`**：经验够就升级 + 全属性提升 + HP 满
- **return "won"/"lost"/"ran"**：调用方根据结果分支

## 四、主程序（5 分钟）

```python
def main():
    print("=== RPG Adventure ===")
    battles = 0

    while player["hp"] > 0:
        enemy = pick_enemy()
        result = battle(player, enemy)
        if result == "won":
            battles += 1
        elif result == "lost":
            break

    print(f"\n=== Game Over ===")
    print(f"Battles won: {battles}, Level: {player['level']}")

main()
```
